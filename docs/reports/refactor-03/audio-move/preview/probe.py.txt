class Addresses(ast.NodeTransformer):
    def visit_ImportFrom(self, node):
        if node.module == 'tools.common.paths':
            return None
        if node.module and node.module.startswith('tools.assets.audio.'):
            node.module = node.module.rsplit('.', 1)[-1]
        return node

    def visit_Call(self, node):
        if ast.unparse(node) == 'project_root(__file__)':
            return ast.parse('Path(__file__).resolve().parents[1]', mode='eval').body
        return self.generic_visit(node)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def candidate(name, source):
    for target in NAMES:
        source = source.replace('from ' + target + ' import ', 'from tools.assets.audio.' + target + ' import ')
    if name == 'audio_composer':
        source = source.replace('ROOT = Path(__file__).resolve().parents[1]',
                                'from tools.common.paths import project_root\n\nROOT = project_root(__file__)')
    return source
