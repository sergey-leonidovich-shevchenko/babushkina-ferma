## Перемещает героя безопасными подшагами; необязательная помощь краю разрешена только для ручной ходьбы.
static func move(game: Node, motion: Vector2, corner_assist: bool = false) -> bool:
	var cost_started_0 := Time.get_ticks_usec()
	var cost_value_0 = game.DebugOverlaySystem.noclip(game)
	Cost.record("navigation_system.gd:11", Time.get_ticks_usec() - cost_started_0)
	if cost_value_0:
		game.player += motion
		return false
	var step_count := maxi(1, ceili(motion.length() / 8.0))
	var step := motion / float(step_count)
	var was_blocked := false
	var assisted := false
	for _index in step_count:
		var before: Vector2 = game.player
		var horizontal: Vector2 = game.player + Vector2(step.x, 0.0)
		var cost_started_1 := Time.get_ticks_usec()
		var cost_value_1 = not is_zero_approx(step.x) and player_step_walkable(game, horizontal)
		Cost.record("navigation_system.gd:21", Time.get_ticks_usec() - cost_started_1)
		if cost_value_1:
			game.player = horizontal
		elif not is_zero_approx(step.x):
			was_blocked = true
		var vertical: Vector2 = game.player + Vector2(0.0, step.y)
		var cost_started_2 := Time.get_ticks_usec()
		var cost_value_2 = not is_zero_approx(step.y) and player_step_walkable(game, vertical)
		Cost.record("navigation_system.gd:26", Time.get_ticks_usec() - cost_started_2)
		if cost_value_2:
			game.player = vertical
		elif not is_zero_approx(step.y):
			was_blocked = true
		var cost_started_3 := Time.get_ticks_usec()
		var cost_value_3 = corner_assist and game.player.is_equal_approx(before)
		Cost.record("navigation_system.gd:30", Time.get_ticks_usec() - cost_started_3)
		if cost_value_3:
			var correction := CornerAssist.resolve(game, step)
			var cost_started_4 := Time.get_ticks_usec()
			var cost_value_4 = not correction.is_zero_approx()
			Cost.record("navigation_system.gd:32", Time.get_ticks_usec() - cost_started_4)
			if cost_value_4:
				game.player += correction
				assisted = true
	if was_blocked:
		game.notify_tutorial("collision")
	return assisted

## Проверяет заявленное методом условие без изменения игрового состояния.
static func is_walkable(game: Node, position: Vector2) -> bool:
	return walkability_reason(game, position) == "walkable"

## Проверяет физическое основание; пустой профиль декора не превращается в невидимую круглую стену.
static func circle_intersects_rect(center: Vector2, radius: float, rect: Rect2) -> bool:
	var cost_started_47 := Time.get_ticks_usec()
	var cost_value_47 = not rect.has_area()
	Cost.record("navigation_system.gd:165", Time.get_ticks_usec() - cost_started_47)
	if cost_value_47:
		return false
	var closest := Vector2(clampf(center.x, rect.position.x, rect.end.x), clampf(center.y, rect.position.y, rect.end.y))
	return center.distance_squared_to(closest) < radius * radius


## Разрешает выход из уже возникшего пересечения врага, но не вход в него и не проход сквозь стены.
static func player_step_walkable(game: Node, position: Vector2) -> bool:
	var cost_started_48 := Time.get_ticks_usec()
	var cost_value_48 = is_walkable(game, position)
	Cost.record("navigation_system.gd:173", Time.get_ticks_usec() - cost_started_48)
	if cost_value_48:
		return true
	var cost_started_49 := Time.get_ticks_usec()
	var cost_value_49 = walkability_reason(game, position, -1, false) != "walkable"
	Cost.record("navigation_system.gd:175", Time.get_ticks_usec() - cost_started_49)
	if cost_value_49:
		return false
	var current_depth := enemy_penetration(game, game.player)
	return current_depth > 0.0 and enemy_penetration(game, position) < current_depth - 0.001


## Измеряет суммарную глубину наложения героя на живых врагов для монотонного выхода наружу.
static func enemy_penetration(game: Node, position: Vector2) -> float:
	var depth := 0.0
	if game.current_location == "overworld" and game.slime_alive:
		depth += maxf(0.0, game.PLAYER_RADIUS + 28.0 - position.distance_to(game.slime_position))
	for enemy in game.enemy_nodes:
		if enemy.alive and enemy.location == game.current_location:
			depth += maxf(0.0, game.PLAYER_RADIUS + 30.0 - position.distance_to(enemy.position))
	if game.current_location == "cave_sanctum":
		for minion in game.state.world.cave_boss.minions:
			var cost_started_50 := Time.get_ticks_usec()
			var cost_value_50 = minion.alive and float(minion.emerge) <= 0.0
			Cost.record("navigation_system.gd:191", Time.get_ticks_usec() - cost_started_50)
			if cost_value_50:
				depth += maxf(0.0, game.PLAYER_RADIUS + 22.0 - position.distance_to(minion.position))
	return depth


## Перемещает мобильного врага по двум осям, позволяя скользить вдоль препятствий.
static func move_enemy(game: Node, enemy_index: int, motion: Vector2) -> Vector2:
	var cost_started_51 := Time.get_ticks_usec()
	var cost_value_51 = enemy_index < 0 or enemy_index >= game.enemy_nodes.size()
	Cost.record("navigation_system.gd:198", Time.get_ticks_usec() - cost_started_51)
	if cost_value_51:
		return Vector2.ZERO
	var position: Vector2 = game.enemy_nodes[enemy_index].position
	var steps := maxi(1, ceili(motion.length() / 6.0))
	var step := motion / float(steps)
	for _index in steps:
		var horizontal := position + Vector2(step.x, 0.0)
		var cost_started_52 := Time.get_ticks_usec()
		var cost_value_52 = not is_zero_approx(step.x) and enemy_position_walkable(game, horizontal, enemy_index, position)
		Cost.record("navigation_system.gd:205", Time.get_ticks_usec() - cost_started_52)
		if cost_value_52:
			position = horizontal
		var vertical := position + Vector2(0.0, step.y)
		var cost_started_53 := Time.get_ticks_usec()
		var cost_value_53 = not is_zero_approx(step.y) and enemy_position_walkable(game, vertical, enemy_index, position)
		Cost.record("navigation_system.gd:208", Time.get_ticks_usec() - cost_started_53)
		if cost_value_53:
			position = vertical
	return position


## Проверяет путь врага без ложного столкновения с ним самим.
static func enemy_position_walkable(game: Node, position: Vector2, enemy_index: int, previous: Vector2 = Vector2.INF) -> bool:
	const RADIUS := 27.0
	var cost_started_54 := Time.get_ticks_usec()
